function loadImage(src, cacheKey = src) {
  if (imageCache.has(cacheKey)) {
    const cached = imageCache.get(cacheKey);
    if (cached.complete && cached.naturalWidth > 0) return Promise.resolve(cached);
  }
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.decoding = "async";
    img.onload = () => {
      imageCache.set(cacheKey, img);
      resolve(img);
    };
    img.onerror = () => reject(new Error(`Failed to load image: ${cacheKey}`));
    img.src = src;
  });
}

function fitDimensions(w, h, maxEdge) {
  const longest = Math.max(w, h);
  const scale = longest > maxEdge ? maxEdge / longest : 1;
  return { width: Math.round(w * scale), height: Math.round(h * scale) };
}

function evenDimensions(w, h) {
  return { width: w - (w % 2), height: h - (h % 2) };
}

async function loadPhotoFromFile(file) {
  const sourceUrl = URL.createObjectURL(file);
  let source;
  try {
    source = await loadImage(sourceUrl, sourceUrl);
  } finally {
    URL.revokeObjectURL(sourceUrl);
  }

  const naturalW = source.naturalWidth || source.width;
  const naturalH = source.naturalHeight || source.height;
  const size = fitDimensions(naturalW, naturalH, CANVAS.workingMaxEdge);

  const canvas = document.createElement("canvas");
  canvas.width = size.width;
  canvas.height = size.height;
  const ctx = canvas.getContext("2d");
  ctx.imageSmoothingQuality = "high";
  ctx.drawImage(source, 0, 0, size.width, size.height);

  const blob = await new Promise((resolve, reject) => {
    canvas.toBlob((b) => b ? resolve(b) : reject(new Error("toBlob failed")), "image/png");
  });
  const url = URL.createObjectURL(blob);
  const image = await loadImage(url, url);
  return { image, url, width: size.width, height: size.height };
}

async function openCamera() {
  state.photoError = "";
  const canTryLiveCamera = navigator.mediaDevices?.getUserMedia && allowsBrowserFeature("camera") && !isEmbeddedPage();
  if (!canTryLiveCamera) {
    fallbackToCameraPicker();
    return;
  }

  let stream = null;
  let facing = "environment";

  function close() {
    if (stream) stream.getTracks().forEach((track) => track.stop());
    stream = null;
    modalRoot.innerHTML = "";
  }

  async function start() {
    const video = modalRoot.querySelector("video");
    const loading = modalRoot.querySelector(".camera-loading");
    if (stream) stream.getTracks().forEach((track) => track.stop());
    loading.hidden = false;
    try {
      stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: facing, width: { ideal: 1280 }, height: { ideal: 960 } },
        audio: false,
      });
      video.srcObject = stream;
      video.style.transform = facing === "user" ? "scaleX(-1)" : "";
      await video.play().catch(() => {});
      loading.hidden = true;
    } catch (err) {
      close();
      fallbackToCameraPicker();
      state.photoError =
        err?.name === "SecurityError" || err?.name === "NotAllowedError"
          ? "Camera access was blocked or denied. Choose a photo from your device instead."
          : "We couldn't open the live camera. Choose a photo from your device instead.";
      render();
    }
  }

  modalRoot.innerHTML = `
    <div class="camera-modal" role="dialog" aria-modal="true" aria-label="Take a photo">
      <div class="camera-box">
        <video playsinline muted></video>
        <div class="camera-loading">Starting camera...</div>
      </div>
      <div class="camera-actions">
        <button class="button secondary" type="button" id="camera-cancel">Cancel</button>
        <button class="snap-button" type="button" id="camera-snap" aria-label="Take photo"><span></span></button>
        <button class="button secondary" type="button" id="camera-flip">Flip</button>
      </div>
      <p class="muted" style="color:#fff">Tap the orange button to snap.</p>
    </div>
  `;

  modalRoot.querySelector("#camera-cancel").addEventListener("click", close);
  modalRoot.querySelector("#camera-flip").addEventListener("click", async () => {
    facing = facing === "user" ? "environment" : "user";
    await start();
  });
  modalRoot.querySelector("#camera-snap").addEventListener("click", async () => {
    const video = modalRoot.querySelector("video");
    if (!video.videoWidth) return;
    const canvas = document.createElement("canvas");
    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;
    const ctx = canvas.getContext("2d");
    if (facing === "user") {
      ctx.translate(canvas.width, 0);
      ctx.scale(-1, 1);
    }
    ctx.drawImage(video, 0, 0);
    const blob = await new Promise((resolve) => canvas.toBlob(resolve, "image/png"));
    close();
    if (blob) await handlePhotoFile(new File([blob], "snapshot.png", { type: "image/png" }));
  });

  await start();
}

function midpointNorm(a, b) {
  return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
}

function isStraight(line) {
  if (!line.mid) return true;
  const m = midpointNorm(line.a, line.b);
  return Math.hypot(line.mid.x - m.x, line.mid.y - m.y) < 0.01;
}

function clientToNorm(element, x, y) {
  const rect = element.getBoundingClientRect();
  return {
    x: clamp((x - rect.left) / rect.width, 0, 1),
    y: clamp((y - rect.top) / rect.height, 0, 1),
  };
}

function hitMouthHandle(x, y) {
  const overlay = document.querySelector("#mouth-overlay");
  const rect = overlay.getBoundingClientRect();
  const points = [
    { h: "a", p: state.line.a },
    { h: "b", p: state.line.b },
    { h: "mid", p: state.line.mid || midpointNorm(state.line.a, state.line.b) },
  ];
  let best = null;
  let bestD = 30;
  for (const item of points) {
    const d = Math.hypot(x - (rect.left + item.p.x * rect.width), y - (rect.top + item.p.y * rect.height));
    if (d <= bestD) {
      bestD = d;
      best = item.h;
    }
  }
  return best;
}

function onMouthDown(event) {
  event.preventDefault();
  document.querySelector("#mouth-hint")?.classList.add("hidden");
  event.currentTarget.setPointerCapture(event.pointerId);
  mouthUi.pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });

  if (mouthUi.pointers.size === 2) {
    const pts = [...mouthUi.pointers.values()];
    mouthUi.gesture = {
      type: "pinch",
      dist0: Math.hypot(pts[1].x - pts[0].x, pts[1].y - pts[0].y) || 1,
      midX0: (pts[0].x + pts[1].x) / 2,
      midY0: (pts[0].y + pts[1].y) / 2,
      s0: mouthUi.view.s,
      tx0: mouthUi.view.tx,
      ty0: mouthUi.view.ty,
    };
    mouthUi.draft = null;
    drawMouthOverlay();
    return;
  }

  if (mouthUi.pointers.size !== 1 || mouthUi.suppress) return;
  const handle = hitMouthHandle(event.clientX, event.clientY);
  if (handle) {
    mouthUi.gesture = { type: "handle", handle };
  } else if (mouthUi.view.s > 1) {
    mouthUi.gesture = {
      type: "pan",
      tx0: mouthUi.view.tx,
      ty0: mouthUi.view.ty,
      x0: event.clientX,
      y0: event.clientY,
    };
  } else {
    const p = clientToNorm(event.currentTarget, event.clientX, event.clientY);
    mouthUi.draft = { a: p, b: p, mid: p };
    mouthUi.gesture = { type: "draw" };
    drawMouthOverlay();
  }
}

function onMouthMove(event) {
  if (!mouthUi.pointers.has(event.pointerId)) return;
  mouthUi.pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });
  const g = mouthUi.gesture;
  const overlay = document.querySelector("#mouth-overlay");

  if (g.type === "pinch" && mouthUi.pointers.size >= 2) {
    const pts = [...mouthUi.pointers.values()];
    const dist = Math.hypot(pts[1].x - pts[0].x, pts[1].y - pts[0].y) || 1;
    const midX = (pts[0].x + pts[1].x) / 2;
    const midY = (pts[0].y + pts[1].y) / 2;
    const rect = document.querySelector("#mouth-stage").getBoundingClientRect();
    const s = clamp(g.s0 * (dist / g.dist0), 1, 4);
    const fx = g.midX0 - rect.left;
    const fy = g.midY0 - rect.top;
    mouthUi.view = clampMouthView(
      fx - ((fx - g.tx0) / g.s0) * s + (midX - g.midX0),
      fy - ((fy - g.ty0) / g.s0) * s + (midY - g.midY0),
      s,
    );
    applyMouthView();
    return;
  }

  if (mouthUi.pointers.size >= 2) return;

  if (g.type === "draw" && mouthUi.draft) {
    mouthUi.draft.b = clientToNorm(overlay, event.clientX, event.clientY);
    mouthUi.draft.mid = midpointNorm(mouthUi.draft.a, mouthUi.draft.b);
    drawMouthOverlay();
  } else if (g.type === "handle") {
    const p = clientToNorm(overlay, event.clientX, event.clientY);
    if (g.handle === "mid") {
      state.line.mid = p;
    } else {
      state.line[g.handle] = p;
      if (isStraight(state.line)) state.line.mid = midpointNorm(state.line.a, state.line.b);
    }
    drawMouthOverlay();
  } else if (g.type === "pan") {
    mouthUi.view = clampMouthView(g.tx0 + event.clientX - g.x0, g.ty0 + event.clientY - g.y0, mouthUi.view.s);
    applyMouthView();
  }
}

function onMouthUp(event) {
  const wasPinch = mouthUi.gesture.type === "pinch";
  mouthUi.pointers.delete(event.pointerId);

  if (mouthUi.gesture.type === "draw" && mouthUi.pointers.size === 0 && mouthUi.draft) {
    const d = mouthUi.draft;
    const rect = document.querySelector("#mouth-overlay").getBoundingClientRect();
    const px = Math.hypot((d.b.x - d.a.x) * rect.width, (d.b.y - d.a.y) * rect.height);
    if (px >= 18) state.line = { a: d.a, b: d.b, mid: midpointNorm(d.a, d.b) };
    mouthUi.draft = null;
    drawMouthOverlay();
  }

  if (mouthUi.pointers.size === 0) {
    mouthUi.gesture = { type: "none" };
    mouthUi.suppress = false;
  } else if (wasPinch) {
    mouthUi.gesture = { type: "none" };
    mouthUi.suppress = true;
  }
}

function clampMouthView(tx, ty, s) {
  if (s <= 1) return { s: 1, tx: 0, ty: 0 };
  const rect = document.querySelector("#mouth-stage").getBoundingClientRect();
  return {
    s,
    tx: clamp(tx, rect.width * (1 - s), 0),
    ty: clamp(ty, rect.height * (1 - s), 0),
  };
}

function zoomAbout(cx, cy, scale) {
  const rect = document.querySelector("#mouth-stage").getBoundingClientRect();
  const s0 = mouthUi.view.s;
  const s = clamp(scale, 1, 4);
  const px = cx - rect.left;
  const py = cy - rect.top;
  mouthUi.view = clampMouthView(
    px - ((px - mouthUi.view.tx) / s0) * s,
    py - ((py - mouthUi.view.ty) / s0) * s,
    s,
  );
  applyMouthView();
}

function zoomButton(factor) {
  const rect = document.querySelector("#mouth-stage").getBoundingClientRect();
  zoomAbout(rect.left + rect.width / 2, rect.top + rect.height / 2, mouthUi.view.s * factor);
}

function applyMouthView() {
  const zoomStage = document.querySelector("#zoom-stage");
  if (!zoomStage) return;
  zoomStage.style.transform = `translate(${mouthUi.view.tx}px, ${mouthUi.view.ty}px) scale(${mouthUi.view.s})`;
  drawMouthOverlay();
}

function mouthPath(line) {
  const a = line.a;
  const b = line.b;
  const m = line.mid || midpointNorm(a, b);
  const cx = (2 * m.x - (a.x + b.x) / 2) * 100;
  const cy = (2 * m.y - (a.y + b.y) / 2) * 100;
  return `M ${a.x * 100} ${a.y * 100} Q ${cx} ${cy} ${b.x * 100} ${b.y * 100}`;
}

function drawMouthOverlay() {
  const overlay = document.querySelector("#mouth-overlay");
  if (!overlay || !state.line) return;
  const guideScale = 1 / mouthUi.view.s;
  const shown = mouthUi.draft || state.line;
  const pts = [
    { h: "a", p: state.line.a, mid: false },
    { h: "b", p: state.line.b, mid: false },
    { h: "mid", p: state.line.mid || midpointNorm(state.line.a, state.line.b), mid: true },
  ];
  overlay.innerHTML = `
    <svg viewBox="0 0 100 100" preserveAspectRatio="none">
      <path d="${mouthPath(shown)}" fill="none" stroke="#fff" stroke-width="${6 * guideScale}" stroke-linecap="round"></path>
      <path d="${mouthPath(shown)}" fill="none" stroke="#ec4899" stroke-width="${3 * guideScale}" stroke-dasharray="${1 * guideScale} ${3 * guideScale}" stroke-linecap="round"></path>
    </svg>
    ${pts.map((item) => `
      <span class="handle ${item.mid ? "mid" : ""}" style="left:${item.p.x * 100}%;top:${item.p.y * 100}%;transform:translate(-50%,-50%) scale(${guideScale})">
        ${item.mid ? `<span class="curve-label">curve</span>` : ""}
      </span>
    `).join("")}
  `;
}

function getAudioContext() {
  if (!sharedAudioContext) {
    const Ctor = window.AudioContext || window.webkitAudioContext;
    sharedAudioContext = new Ctor();
  }
  if (sharedAudioContext.state === "suspended") sharedAudioContext.resume();
  return sharedAudioContext;
}

function pickAudioMime() {
  const candidates = [
    "audio/webm;codecs=opus",
    "audio/webm",
    "audio/mp4;codecs=mp4a.40.2",
    "audio/mp4",
    "audio/ogg;codecs=opus",
  ];
  if (typeof MediaRecorder === "undefined") return "";
  return candidates.find((candidate) => MediaRecorder.isTypeSupported(candidate)) || "";
}

async function startRecording() {
  state.recordError = "";
  if (!navigator.mediaDevices?.getUserMedia) {
    state.recordStatus = "denied";
    state.recordError = "This browser does not have microphone recording. Upload audio instead, or try opening the file in Chrome, Edge, or Safari.";
    renderStep3();
    return;
  }
  if (!allowsBrowserFeature("microphone")) {
    state.recordStatus = "denied";
    state.recordError = permissionPolicyMessage("microphone");
    renderStep3();
    return;
  }
  state.recordStatus = "processing";
  renderStep3();
  try {
    const audioContext = getAudioContext();
    recordStream = await navigator.mediaDevices.getUserMedia({ audio: true });
    const mime = pickAudioMime();
    recorder = new MediaRecorder(recordStream, mime ? { mimeType: mime } : undefined);
    recordChunks = [];
    recorder.ondataavailable = (event) => {
      if (event.data.size) recordChunks.push(event.data);
    };
    recorder.onstop = () => finalizeRecording(recorder.mimeType || mime || "audio/webm");

    const src = audioContext.createMediaStreamSource(recordStream);
    recordAnalyser = audioContext.createAnalyser();
    recordAnalyser.fftSize = 1024;
    src.connect(recordAnalyser);

    recorder.start();
    recordStartedAt = performance.now();
    state.recordElapsed = 0;
    state.recordStatus = "recording";
    renderStep3();
    recordRaf = requestAnimationFrame(meterLoop);
    recordTimer = window.setInterval(() => {
      const elapsed = (performance.now() - recordStartedAt) / 1000;
      state.recordElapsed = elapsed;
      updateRecordingUi();
      if (elapsed >= AUDIO.maxSeconds) stopRecording();
    }, 100);
  } catch (err) {
    cleanupRecording();
    state.recordStatus = "denied";
    state.recordError =
      err?.name === "SecurityError" || err?.name === "NotAllowedError"
        ? "Microphone access was blocked or denied. Open this HTML file directly in a browser tab and allow microphone access, or upload audio instead."
        : "TalkLine needs your microphone to record. Allow microphone access in your browser, then try again, or upload audio instead.";
    renderStep3();
  }
}

async function handleAudioFile(file) {
  state.recordStatus = "processing";
  state.recordError = "";
  state.recordPlaying = false;
  renderStep3();
  try {
    const buffer = await decodeAudio(file);
    const url = URL.createObjectURL(file);
    if (state.audio?.url) URL.revokeObjectURL(state.audio.url);
    state.audio = {
      blob: file,
      url,
      buffer,
      durationSec: Math.min(buffer.duration, AUDIO.maxSeconds),
      mimeType: file.type || "audio/*",
    };
    state.envelope = computeEnvelope(buffer);
    state.recordStatus = "idle";
  } catch {
    state.recordStatus = "error";
    state.recordError = "That audio file could not be opened. Try a WAV, MP3, M4A, OGG, or WebM file.";
  }
  renderStep3();
}

function meterLoop() {
  if (!recordAnalyser) return;
  const buf = new Float32Array(recordAnalyser.fftSize);
  recordAnalyser.getFloatTimeDomainData(buf);
  let sum = 0;
  for (let i = 0; i < buf.length; i++) sum += buf[i] * buf[i];
  const rms = Math.sqrt(sum / buf.length);
  state.recordLevel = state.recordLevel * 0.6 + Math.min(1, rms * 4) * 0.4;
  updateRecordingUi();
  recordRaf = requestAnimationFrame(meterLoop);
}

function updateRecordingUi() {
  const timer = document.querySelector("#record-timer");
  const ring = document.querySelector("#record-ring");
  if (timer && state.recordStatus === "recording") {
    const remaining = Math.max(0, AUDIO.maxSeconds - state.recordElapsed);
    timer.innerHTML = `${state.recordElapsed.toFixed(1)}s <span class="muted">${remaining.toFixed(0)}s left</span>`;
  }
  if (ring) {
    const size = 10.5 + state.recordLevel * 3.8;
    ring.style.width = `${size}rem`;
    ring.style.height = `${size}rem`;
  }
}

function stopRecording() {
  if (recorder && recorder.state !== "inactive") {
    state.recordStatus = "processing";
    recorder.stop();
  }
  cleanupRecording();
  renderStep3();
}

function cleanupRecording() {
  cancelAnimationFrame(recordRaf);
  if (recordTimer) window.clearInterval(recordTimer);
  recordTimer = 0;
  if (recordStream) recordStream.getTracks().forEach((track) => track.stop());
  recordStream = null;
  recordAnalyser = null;
  state.recordLevel = 0;
}

async function finalizeRecording(mimeType) {
  try {
    const blob = new Blob(recordChunks, { type: mimeType });
    const buffer = await decodeAudio(blob);
    const url = URL.createObjectURL(blob);
    if (state.audio?.url) URL.revokeObjectURL(state.audio.url);
    state.audio = {
      blob,
      url,
      buffer,
      durationSec: Math.min(buffer.duration, AUDIO.maxSeconds),
      mimeType,
    };
    state.envelope = computeEnvelope(buffer);
    state.recordStatus = "idle";
  } catch {
    state.recordStatus = "error";
  }
  renderStep3();
}

async function decodeAudio(blob) {
  const arrayBuffer = await blob.arrayBuffer();
  return await getAudioContext().decodeAudioData(arrayBuffer.slice(0));
}

function clearAudio() {
  if (state.audio?.url) URL.revokeObjectURL(state.audio.url);
  state.audio = null;
  state.envelope = null;
  state.recordPlaying = false;
  state.recordError = "";
  state.recordStatus = "idle";
  renderStep3();
}

function toMono(buffer) {
  const channels = buffer.numberOfChannels;
  if (channels === 1) return buffer.getChannelData(0);
  const len = buffer.length;
  const out = new Float32Array(len);
  for (let c = 0; c < channels; c++) {
    const data = buffer.getChannelData(c);
    for (let i = 0; i < len; i++) out[i] += data[i];
  }
  for (let i = 0; i < len; i++) out[i] /= channels;
  return out;
}

function computeEnvelope(buffer) {
  const mono = toMono(buffer);
  const frameCount = Math.max(1, Math.ceil(buffer.duration * ANIMATION.fps));
  const windowSize = Math.max(1, Math.floor(buffer.sampleRate / ANIMATION.fps));
  const rms = new Float32Array(frameCount);

  for (let f = 0; f < frameCount; f++) {
    const start = f * windowSize;
    const end = Math.min(start + windowSize, mono.length);
    let sum = 0;
    for (let i = start; i < end; i++) sum += mono[i] * mono[i];
    rms[f] = Math.sqrt(sum / Math.max(1, end - start));
  }

  const sorted = Float32Array.from(rms).sort();
  const pIdx = clamp(Math.floor(ANIMATION.normalizePercentile * (sorted.length - 1)), 0, sorted.length - 1);
  const ref = Math.max(sorted[pIdx], 1e-4);
  const env = new Float32Array(frameCount);
  let prev = 0;

  for (let f = 0; f < frameCount; f++) {
    let v = clamp(rms[f] / ref, 0, 1);
    v = Math.pow(v, ANIMATION.gamma);
    let target = 0;
    if (v >= ANIMATION.silenceGate) {
      const ranged = (v - ANIMATION.silenceGate) / (1 - ANIMATION.silenceGate);
      target = ANIMATION.openFloor + (1 - ANIMATION.openFloor) * ranged;
    }
    const coeff = target > prev ? ANIMATION.attack : ANIMATION.release;
    prev = prev + coeff * (target - prev);
    env[f] = clamp(prev, 0, 1);
  }
  return env;
}

function envelopeAt(env, timeSec) {
  if (!env?.length) return 0;
  const idx = clamp(Math.floor(timeSec * ANIMATION.fps), 0, env.length - 1);
  return env[idx];
}

  function move(event) {
    if (!gesture) return;
    const sticker = state.stickers.find((s) => s.id === id);
    if (!sticker) return;
    sticker.x = clamp(gesture.startX + (event.clientX - gesture.x0) / gesture.rect.width, 0, 1);
    sticker.y = clamp(gesture.startY + (event.clientY - gesture.y0) / gesture.rect.height, 0, 1);
    activeElement.style.left = `${sticker.x * 100}%`;
    activeElement.style.top = `${sticker.y * 100}%`;
    renderStickersCacheKey = "";
  }

  function end() {
    gesture = null;
  }
}

function bindStickerResize(button) {
  let gesture = null;
  const id = button.dataset.resize;
  button.addEventListener("pointerdown", (event) => {
    event.stopPropagation();
    const sticker = state.stickers.find((s) => s.id === id);
    if (!sticker) return;
    state.selectedSticker = id;
    button.setPointerCapture(event.pointerId);
    const rect = document.querySelector("#sticker-stage").getBoundingClientRect();
    const cx = rect.left + sticker.x * rect.width;
    const cy = rect.top + sticker.y * rect.height;
    gesture = {
      startScale: sticker.scale,
      startRot: sticker.rot,
      startDist: Math.hypot(event.clientX - cx, event.clientY - cy) || 1,
      startAngle: Math.atan2(event.clientY - cy, event.clientX - cx),
      rect,
      cx,
      cy,
    };
  });
  button.addEventListener("pointermove", (event) => {
    if (!gesture) return;
    const dist = Math.hypot(event.clientX - gesture.cx, event.clientY - gesture.cy) || 1;
    const angle = Math.atan2(event.clientY - gesture.cy, event.clientX - gesture.cx);
    const sticker = state.stickers.find((s) => s.id === id);
    if (!sticker) return;
    sticker.scale = clamp(gesture.startScale * (dist / gesture.startDist), 0.06, 1.4);
    sticker.rot = gesture.startRot + (angle - gesture.startAngle);
    renderStickersCacheKey = "";
    const item = document.querySelector(`[data-sticker-id="${id}"]`);
    if (item) {
      item.style.width = `${sticker.scale * 100}%`;
      item.style.transform = `translate(-50%,-50%) rotate(${sticker.rot}rad)`;
      item.querySelectorAll(".sticker-tool").forEach((tool) => {
        tool.style.transform = `rotate(${-sticker.rot}rad)`;
      });
    }
  });
  button.addEventListener("pointerup", () => {
    gesture = null;
  });
  button.addEventListener("pointercancel", () => {
    gesture = null;
  });
}

async function exportVideo() {
  if (!canExport()) return;
  stopPreview();
  exportAbort = false;
  if (state.resultUrl) URL.revokeObjectURL(state.resultUrl);
  state.resultUrl = "";
  state.resultFilename = "";
  state.resultType = "";
  state.exportState = { phase: "rendering", progress: 0, message: "Getting ready..." };
  renderStep5();

  try {
    const blob = await exportWithMediaRecorder((progress, message) => {
      state.exportState = { phase: "rendering", progress, message };
      const fill = document.querySelector(".progress-fill");
      const head = document.querySelector(".progress-head");
      if (fill) fill.style.width = `${Math.max(3, progress * 100)}%`;
      if (head) head.innerHTML = `<span>${escapeHtml(message)}</span><span>${Math.round(progress * 100)}%</span>`;
    });
    const url = URL.createObjectURL(blob);
    const isMp4 = blob.type.includes("mp4");
    state.resultUrl = url;
    state.resultFilename = `talking-photo.${isMp4 ? "mp4" : "webm"}`;
    state.resultType = isMp4 ? "mp4" : "webm";
    state.exportState = { phase: "done", progress: 1, message: "Saved!" };
    triggerDownload(url, state.resultFilename);
  } catch (err) {
    if (err?.name === "AbortError") {
      state.exportState = { phase: "idle", progress: 0, message: "" };
    } else {
      state.exportState = {
        phase: "error",
        progress: 0,
        message: "",
        error: "Something went wrong making your video. Try again.",
      };
    }
  }
  renderStep5();
}

async function exportWithMediaRecorder(onProgress) {
  const fitted = fitDimensions(state.photo.width, state.photo.height, CANVAS.exportMaxEdge);
  const size = evenDimensions(fitted.width, fitted.height);
  const canvas = document.createElement("canvas");
  canvas.width = size.width;
  canvas.height = size.height;
  const ctx = canvas.getContext("2d", { alpha: false });
  const stickers = await getRenderStickers();
  const scene = {
    photo: state.photo.image,
    width: size.width,
    height: size.height,
    line: state.line,
    stickers,
  };
  const audioContext = getAudioContext();
  await audioContext.resume();
  const dest = audioContext.createMediaStreamDestination();
  const source = audioContext.createBufferSource();
  source.buffer = state.audio.buffer;
  source.connect(dest);

  const canvasStream = canvas.captureStream(ANIMATION.fps);
  const stream = new MediaStream([...canvasStream.getVideoTracks(), ...dest.stream.getAudioTracks()]);
  const mime = [
    'video/mp4;codecs="avc1.42E01E, mp4a.40.2"',
    "video/mp4",
    "video/webm;codecs=vp9,opus",
    "video/webm;codecs=vp8,opus",
    "video/webm",
  ].find((type) => MediaRecorder.isTypeSupported(type)) || "video/webm";
  const mediaRecorder = new MediaRecorder(stream, { mimeType: mime });
  const chunks = [];
  mediaRecorder.ondataavailable = (event) => {
    if (event.data.size) chunks.push(event.data);
  };

  return await new Promise((resolve, reject) => {
    let raf = 0;
    let stopped = false;
    const startTime = audioContext.currentTime;
    const duration = Math.min(state.audio.buffer.duration, AUDIO.maxSeconds);

    function finish(error) {
      if (stopped) return;
      stopped = true;
      cancelAnimationFrame(raf);
      stream.getTracks().forEach((track) => track.stop());
      try {
        source.stop();
      } catch {
        /* source may already be stopped */
      }
      if (error) reject(error);
    }

    function tick() {
      if (exportAbort) {
        mediaRecorder.stop();
        finish(new DOMException("Export cancelled", "AbortError"));
        return;
      }
      const elapsed = audioContext.currentTime - startTime;
      drawFrame(ctx, scene, envelopeAt(state.envelope, elapsed));
      onProgress(Math.min(1, elapsed / duration), "Recording your video...");
      if (elapsed >= duration) {
        mediaRecorder.stop();
        return;
      }
      raf = requestAnimationFrame(tick);
    }

    mediaRecorder.onstop = () => {
      if (stopped && exportAbort) return;
      cancelAnimationFrame(raf);
      stream.getTracks().forEach((track) => track.stop());
      resolve(new Blob(chunks, { type: mime }));
    };
    mediaRecorder.onerror = () => {
      finish(new Error("Recording failed"));
    };

    mediaRecorder.start(100);
    source.start();
    tick();
  });
}

function toPx(point, w, h) {
  return { x: point.x * w, y: point.y * h };
}

function midpoint(a, b) {
  return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
}

function dist(a, b) {
  return Math.hypot(a.x - b.x, a.y - b.y);
}

function hingeGeometry(line, w, h) {
  const a = toPx(line.a, w, h);
  const b = toPx(line.b, w, h);
  const pivot = midpoint(a, b);
  const len = Math.max(dist(a, b), 1e-3);
  const dir = { x: (b.x - a.x) / len, y: (b.y - a.y) / len };
  let normal = { x: -dir.y, y: dir.x };
  if (normal.y < 0) normal = { x: -normal.x, y: -normal.y };
  return { a, b, pivot, dir, normal, len };
}

function sampleMouthCurve(line, w, h, samples = 16) {
  const a = toPx(line.a, w, h);
  const b = toPx(line.b, w, h);
  const midPoint = line.mid ? toPx(line.mid, w, h) : midpoint(a, b);
  const control = { x: 2 * midPoint.x - (a.x + b.x) / 2, y: 2 * midPoint.y - (a.y + b.y) / 2 };
  const pts = [];
  for (let i = 0; i <= samples; i++) {
    const t = i / samples;
    const u = 1 - t;
    pts.push({
      x: u * u * a.x + 2 * u * t * control.x + t * t * b.x,
      y: u * u * a.y + 2 * u * t * control.y + t * t * b.y,
    });
  }
  return pts;
}

function jawPose(openAmount, lineLenPx) {
  const a = clamp(openAmount, 0, 1);
  return {
    drop: a * ANIMATION.maxDropFraction * lineLenPx,
    angle: (a * ANIMATION.maxAngleDeg * Math.PI) / 180,
  };
}

function drawFrame(ctx, scene, openAmount) {
  const { photo, width: w, height: h, line } = scene;
  ctx.clearRect(0, 0, w, h);
  ctx.drawImage(photo, 0, 0, w, h);

  if (line) {
    const { a, b, pivot, dir, normal, len } = hingeGeometry(line, w, h);
    const { drop, angle } = jawPose(openAmount, len);

    if (drop > 0.5) {
      const samples = 16;
      const pts = sampleMouthCurve(line, w, h, samples);
      const grad = ctx.createLinearGradient(pivot.x, pivot.y, pivot.x + normal.x * drop * 2, pivot.y + normal.y * drop * 2);
      grad.addColorStop(0, "#150a0d");
      grad.addColorStop(0.5, "#2c0f13");
      grad.addColorStop(1, "#3d1117");

      ctx.beginPath();
      ctx.moveTo(pts[0].x, pts[0].y);
      for (let i = 1; i <= samples; i++) ctx.lineTo(pts[i].x, pts[i].y);
      for (let i = samples; i >= 0; i--) {
        const t = i / samples;
        const droop = Math.sin(t * Math.PI) * drop * 0.6;
        ctx.lineTo(pts[i].x + normal.x * (drop + droop), pts[i].y + normal.y * (drop + droop));
      }
      ctx.closePath();
      ctx.fillStyle = grad;
      ctx.fill();

      const chinDrop = len * 0.95;
      ctx.save();
      ctx.translate(normal.x * drop, normal.y * drop);
      ctx.translate(pivot.x, pivot.y);
      ctx.rotate(ANIMATION.rotationSign * angle);
      ctx.translate(-pivot.x, -pivot.y);
      ctx.beginPath();
      ctx.moveTo(pts[0].x, pts[0].y);
      for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].x, pts[i].y);
      ctx.bezierCurveTo(
        b.x + dir.x * len * 0.15 + normal.x * chinDrop,
        b.y + dir.y * len * 0.15 + normal.y * chinDrop,
        a.x - dir.x * len * 0.15 + normal.x * chinDrop,
        a.y - dir.y * len * 0.15 + normal.y * chinDrop,
        a.x,
        a.y,
      );
      ctx.closePath();
      ctx.clip();
      ctx.drawImage(photo, 0, 0, w, h);
      ctx.beginPath();
      ctx.moveTo(pts[0].x, pts[0].y);
      for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].x, pts[i].y);
      ctx.strokeStyle = `rgba(0,0,0,${Math.min(0.5, (drop / len) * 1.6)})`;
      ctx.lineWidth = Math.max(2, len * 0.05);
      ctx.lineCap = "round";
      ctx.stroke();
      ctx.restore();
    }
  }

  drawStickers(ctx, scene);
}

function drawStickers(ctx, scene) {
  for (const sticker of scene.stickers || []) {
    const { placement, image, aspect } = sticker;
    const boxW = placement.scale * scene.width;
    const boxH = boxW / (aspect || 1);
    const cx = placement.x * scene.width;
    const cy = placement.y * scene.height;
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(placement.rot);
    ctx.drawImage(image, -boxW / 2, -boxH / 2, boxW, boxH);
    ctx.restore();
  }
}

async function getRenderStickers() {
  const key = JSON.stringify(state.stickers.map((s) => [s.id, s.key, s.src || "", s.x, s.y, s.scale, s.rot]));
  if (key === renderStickersCacheKey) return renderStickersCache;
  renderStickersCache = await Promise.all(state.stickers.map(async (placement) => {
    const src = stickerSrc(placement);
    const image = await loadImage(src, placement.type === "custom" ? src : `builtin:${placement.key}`);
    const aspect = (image.naturalWidth || image.width || 1) / (image.naturalHeight || image.height || 1);
    return { placement, image, aspect };
  }));
  renderStickersCacheKey = key;
  return renderStickersCache;
}
